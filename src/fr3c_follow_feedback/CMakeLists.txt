cmake_minimum_required(VERSION 3.16)
project(fr3c_follow_feedback)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(Eigen3 REQUIRED)
set(FAIRINO_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../fairino_hardware/libfairino" CACHE PATH "Pinned local SDK 2.3.2")
add_executable(robot_feedback src/robot_feedback.cpp)
target_include_directories(robot_feedback PRIVATE include "${FAIRINO_SDK_ROOT}/include")
target_link_libraries(robot_feedback Eigen3::Eigen "${FAIRINO_SDK_ROOT}/lib/libfairino.so.2.3.2")
ament_target_dependencies(robot_feedback rclcpp geometry_msgs std_msgs tf2_ros)
set_target_properties(robot_feedback PROPERTIES INSTALL_RPATH "${FAIRINO_SDK_ROOT}/lib")
install(TARGETS robot_feedback DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_feedback_helpers test/test_feedback_helpers.cpp)
  target_include_directories(test_feedback_helpers PRIVATE include)
  target_link_libraries(test_feedback_helpers Eigen3::Eigen)
endif()
ament_package()
