cmake_minimum_required(VERSION 3.16)
project(fr3c_follow_driver)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(moveit_core REQUIRED)
find_package(urdf REQUIRED)
find_package(srdfdom REQUIRED)
set(FAIRINO_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../fairino_hardware/libfairino" CACHE PATH "Pinned SDK 2.3.2")
set(FEEDBACK_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../fr3c_follow_feedback/include")
add_library(follow_collision src/collision_guard.cpp)
target_compile_options(follow_collision PRIVATE $<$<NOT:$<CONFIG:Debug>>:-O2>)
target_include_directories(follow_collision PUBLIC include "${FAIRINO_SDK_ROOT}/include")
target_link_libraries(follow_collision Eigen3::Eigen)
ament_target_dependencies(follow_collision moveit_core urdf srdfdom)
foreach(executable servo_probe position_follow rotation_probe pose_follow)
  add_executable(${executable} src/${executable}.cpp)
  target_include_directories(${executable} PRIVATE include "${FEEDBACK_INCLUDE}" "${FAIRINO_SDK_ROOT}/include")
  target_link_libraries(${executable} Eigen3::Eigen "${FAIRINO_SDK_ROOT}/lib/libfairino.so.2.3.2")
  ament_target_dependencies(${executable} rclcpp geometry_msgs std_msgs std_srvs tf2_ros)
  set_target_properties(${executable} PROPERTIES INSTALL_RPATH "${FAIRINO_SDK_ROOT}/lib")
  install(TARGETS ${executable} DESTINATION lib/${PROJECT_NAME})
endforeach()
target_link_libraries(position_follow follow_collision)
target_link_libraries(pose_follow follow_collision)
install(TARGETS follow_collision DESTINATION lib)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_servo_core test/test_servo_core.cpp)
  target_include_directories(test_servo_core PRIVATE include "${FEEDBACK_INCLUDE}" "${FAIRINO_SDK_ROOT}/include")
  target_link_libraries(test_servo_core Eigen3::Eigen)
  ament_add_gtest(test_position_follow test/test_position_follow.cpp)
  target_include_directories(test_position_follow PRIVATE include "${FAIRINO_SDK_ROOT}/include")
  target_link_libraries(test_position_follow Eigen3::Eigen)
  ament_add_gtest(test_rotation_probe test/test_rotation_probe.cpp)
  target_include_directories(test_rotation_probe PRIVATE include "${FAIRINO_SDK_ROOT}/include")
  target_link_libraries(test_rotation_probe Eigen3::Eigen)
  ament_add_gtest(test_native_sdk_boundary test/test_native_sdk_boundary.cpp)
  target_include_directories(test_native_sdk_boundary PRIVATE test/fake_sdk include "${FAIRINO_SDK_ROOT}/include")
  target_link_libraries(test_native_sdk_boundary Eigen3::Eigen)
  ament_add_gtest(test_pose_follow test/test_pose_follow.cpp)
  target_include_directories(test_pose_follow PRIVATE include "${FAIRINO_SDK_ROOT}/include")
  target_link_libraries(test_pose_follow Eigen3::Eigen)
  ament_add_gtest(test_pose_feedback test/test_pose_feedback.cpp)
  target_include_directories(test_pose_feedback PRIVATE include "${FEEDBACK_INCLUDE}" "${FAIRINO_SDK_ROOT}/include")
  target_link_libraries(test_pose_feedback Eigen3::Eigen follow_collision "${FAIRINO_SDK_ROOT}/lib/libfairino.so.2.3.2")
  ament_target_dependencies(test_pose_feedback rclcpp geometry_msgs std_msgs std_srvs tf2_ros)
  set(MODEL_CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/../fairino3mt_v6_moveit2_config/config")
  execute_process(COMMAND /opt/ros/humble/bin/xacro "${MODEL_CONFIG}/fairino3mt_v6_robot.urdf.xacro"
    "initial_positions_file:=${MODEL_CONFIG}/initial_positions.yaml"
    "hardware_plugin:=mock_components/GenericSystem"
    OUTPUT_FILE "${CMAKE_CURRENT_BINARY_DIR}/collision_robot.urdf" COMMAND_ERROR_IS_FATAL ANY)
  ament_add_gtest(test_collision_guard test/test_collision_guard.cpp)
  target_compile_definitions(test_collision_guard PRIVATE COLLISION_URDF="${CMAKE_CURRENT_BINARY_DIR}/collision_robot.urdf"
    COLLISION_SRDF="${MODEL_CONFIG}/fairino3mt_v6_robot.srdf")
  target_link_libraries(test_collision_guard follow_collision)
  ament_add_gtest(test_pose_collision test/test_pose_collision.cpp)
  target_compile_definitions(test_pose_collision PRIVATE COLLISION_URDF="${CMAKE_CURRENT_BINARY_DIR}/collision_robot.urdf"
    COLLISION_SRDF="${MODEL_CONFIG}/fairino3mt_v6_robot.srdf")
  target_link_libraries(test_pose_collision follow_collision)
endif()
ament_package()
